import React, { useState } from 'react';
import { ExamQuestion } from '../data/exam_questions';
import { BookOpen, HelpCircle, ChevronDown, ChevronUp } from 'lucide-react';
import MathRenderer from './MathRenderer';

interface Props {
  question: ExamQuestion;
}

const ExamQuestionCard: React.FC<Props> = ({ question }) => {
  const [isExpanded, setIsExpanded] = useState(false);

  const getSectionColor = (section: string) => {
    switch (section) {
      case 'A': return 'text-duo-blue border-duo-blue';
      case 'B': return 'text-duo-orange border-duo-orange';
      case 'C': return 'text-duo-violet border-duo-violet';
      default: return 'text-slate-400 border-slate-400';
    }
  };

  const getWordLimit = (section: string) => {
    switch (section) {
      case 'A': return 'Max 3 sentences';
      case 'B': return '~100 words';
      case 'C': return '~200 words';
      default: return '';
    }
  };

  return (
    <div className="exam-card glass-panel rounded-2xl mb-4 overflow-hidden border border-white/10 transition-all duration-300 shadow-md">
      <div 
        className="p-5 cursor-pointer hover:bg-white/5 transition-colors"
        onClick={() => setIsExpanded(!isExpanded)}
      >
        <div className="flex justify-between items-start mb-3">
          <div className="flex items-center gap-2">
            <span className={`px-2 py-0.5 rounded-md text-[10px] font-black uppercase tracking-widest border ${getSectionColor(question.section)} bg-white/5`}>
              Section {question.section} ({question.marks} Marks)
            </span>
            <span className="text-xs font-bold text-slate-400 flex items-center gap-1 bg-black/30 px-2 py-0.5 rounded-md">
              <BookOpen className="w-3 h-3" />
              {question.chapter}
            </span>
          </div>
          {isExpanded ? (
            <ChevronUp className="w-5 h-5 text-slate-500" />
          ) : (
            <ChevronDown className="w-5 h-5 text-slate-500" />
          )}
        </div>
        
        <h3 className="text-lg font-bold text-slate-200 leading-snug pr-4">
          <MathRenderer content={question.question} />
        </h3>
      </div>

      {isExpanded && (
        <div className="p-5 pt-2 border-t border-white/5 bg-black/20 animate-in slide-in-from-top-2 duration-300">
           <div className="flex items-start gap-3 mt-2">
             <HelpCircle className="w-5 h-5 text-duo-blue shrink-0 mt-0.5" />
             <div>
               <p className="text-sm font-bold text-slate-300 mb-1">Exam Strategy</p>
               <p className="text-xs text-slate-400 mb-2">
                 Length: <strong className="text-slate-300">{getWordLimit(question.section)}</strong>
               </p>
               <p className="text-xs font-medium text-slate-500 italic">
                 (Model answers for all chapters are unlocked as you progress through the learning path. Use this as a self-check prompt!)
               </p>
             </div>
           </div>
        </div>
      )}
    </div>
  );
};

export default ExamQuestionCard;

import React, { useState, useMemo } from 'react';
import { EXAM_QUESTIONS } from '../data/exam_questions';
import ExamQuestionCard from '../components/ExamQuestionCard';
import { BookOpenCheck, Filter } from 'lucide-react';

const ExamQuestionsPage: React.FC = () => {
  const [filterSection, setFilterSection] = useState<'ALL' | 'A' | 'B' | 'C'>('ALL');

  const filteredQuestions = useMemo(() => {
    if (filterSection === 'ALL') return EXAM_QUESTIONS;
    return EXAM_QUESTIONS.filter(q => q.section === filterSection);
  }, [filterSection]);

  return (
    <div className="flex-grow overflow-y-auto pb-28 pt-6 px-4 no-scrollbar border-x border-white/5 relative">
      <div className="flex items-center justify-center mb-6">
        <BookOpenCheck className="w-8 h-8 text-duo-blue mr-3" />
        <h1 className="text-2xl font-black text-white tracking-tight">University Questions</h1>
      </div>

      <p className="text-center text-slate-400 text-sm font-medium mb-6 px-4">
        Model Question Paper Archive. Tap a question to view exam strategies and expected length.
      </p>

      {/* Filters */}
      <div className="flex justify-center mb-8 sticky top-0 z-20 py-2 glass-panel rounded-2xl mx-2 shadow-lg border border-white/10">
        <div className="flex items-center gap-2 px-2">
          <Filter className="w-4 h-4 text-slate-500 mr-1" />
          {(['ALL', 'A', 'B', 'C'] as const).map(sec => (
            <button
              key={sec}
              onClick={() => setFilterSection(sec)}
              className={`px-3 py-1.5 rounded-xl text-xs font-black uppercase tracking-widest transition-all ${
                filterSection === sec 
                  ? 'bg-duo-blue text-white shadow-md' 
                  : 'text-slate-400 hover:text-slate-200 hover:bg-white/5'
              }`}
            >
              {sec === 'ALL' ? 'All' : `Sec ${sec}`}
            </button>
          ))}
        </div>
      </div>

      {/* Question List */}
      <div className="max-w-md mx-auto">
        {filteredQuestions.length > 0 ? (
          filteredQuestions.map(q => (
            <ExamQuestionCard key={q.id} question={q} />
          ))
        ) : (
          <div className="text-center text-slate-500 py-10">
            No questions found for this filter.
          </div>
        )}
      </div>
    </div>
  );
};

export default ExamQuestionsPage;

import React, { useState, useEffect } from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';
import { UserProgress } from './types';
import Dashboard from './pages/Dashboard';
import LessonPage from './pages/LessonPage';
import ModuleSummaryPage from './pages/ModuleSummaryPage';
import ExamQuestionsPage from './pages/ExamQuestionsPage';
import ModuleNotesPage from './pages/ModuleNotesPage';
import PracticePage from './pages/PracticePage';
import MainLayout from './components/MainLayout';
import { useAnalytics } from './hooks/useAnalytics';
import './App.css';
import './styles/interactive.css';
import './styles/path.css';
import './styles/progress.css';
import './styles/english.css'; 
import './styles/practice.css'; 
import './styles/exam.css';
import './styles/policy.css';

const AnalyticsTracker = () => {
  useAnalytics();
  return null;
};

const App: React.FC = () => {
  const [progress, setProgress] = useState<UserProgress>(() => {
    const saved = localStorage.getItem('duofy_english_vac_progress');
    let initialProgress: UserProgress = {
      completedLessons: [],
      xp: 0,
      currentModuleId: 'module-1-human-rights',
    };

    if (saved) {
      try {
        const parsed = JSON.parse(saved);
        initialProgress = {
          completedLessons: parsed.completedLessons || [],
          xp: parsed.xp || 0,
          currentModuleId: parsed.currentModuleId || 'module-1-human-rights',
        };
      } catch (e) {
        console.error('Failed to parse progress', e);
      }
    }
    return initialProgress;
  });

  const [currentSectionIndex, setCurrentSectionIndex] = useState(0);

  useEffect(() => {
    localStorage.setItem('duofy_english_vac_progress', JSON.stringify(progress));
  }, [progress]);

  useEffect(() => {
    setCurrentSectionIndex(0);
  }, [progress.currentModuleId]);

  const handleModuleChange = (moduleId: string) => {
    setProgress(prev => ({ ...prev, currentModuleId: moduleId }));
    setCurrentSectionIndex(0);
  };

  return (
    <div className="min-h-screen bg-[#0b0f19] text-slate-200 shadow-[0_0_50px_rgba(0,0,0,0.5)] max-w-md mx-auto relative overflow-hidden">
      <BrowserRouter>
        <AnalyticsTracker />
        <Routes>
          <Route element={<MainLayout />}>
            <Route
              path="/"
              element={
                <Dashboard
                  progress={progress}
                  setProgress={setProgress}
                  currentSectionIndex={currentSectionIndex}
                  setCurrentSectionIndex={setCurrentSectionIndex}
                />
              }
            />
            <Route path="/exam" element={<ExamQuestionsPage />} />
            <Route path="/practice" element={<PracticePage />} />
            <Route
              path="/notes"
              element={
                <ModuleNotesPage
                  progress={progress}
                  onModuleChange={handleModuleChange}
                />
              }
            />
          </Route>

          <Route
            path="/lesson/:moduleId/:unitId/:lessonId"
            element={<LessonPage progress={progress} setProgress={setProgress} />}
          />
          <Route path="/summary/:moduleId" element={<ModuleSummaryPage />} />
        </Routes>
      </BrowserRouter>
    </div>
  );
};

export default App;
